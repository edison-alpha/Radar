// Farm economics. One pure function prices a farm plan for the board, board.json, the detail page
// and the plan dialog, so the number on screen is the number in the file.
//
//   slice   = implied value x airdrop share x capital / TVL   (depositors share the launch pro rata)
//   tbill   = capital x T-bill yield x hold days / 365         (what idle dollars would have earned)
//   net     = slice - fees - tbill
//   per hour = net / hands-on hours of the plan
//   verdict: FARM when net > 0; THIN when the slice beats the fees but not the T-bill; SKIP otherwise.
// The airdrop is a scenario, never a forecast: the snapshot records no points programme.
export const ECON = Object.freeze({
 capital: 1000,
 share: 0.1,
 holdDays: 90,
 tbillYield: 0.04,
 capitals: [1000, 10000, 100000],
 shares: [0.05, 0.1, 0.2],
 holds: [30, 90, 180],
});
const cents = (n) => Math.round(n * 100) / 100;
export function farmEconomics(row, options = {}) {
 const capital = options.capital ?? ECON.capital;
 const share = options.share ?? ECON.share;
 const holdDays = options.holdDays ?? ECON.holdDays;
 const fees = row.farm.fees_usd;
 const hours = Math.max(row.farm.minutes, 1) / 60;
 const slice = row.tvl_usd > 0 ? (row.implied_value_usd * share * capital) / row.tvl_usd : 0;
 const tbill = (capital * ECON.tbillYield * holdDays) / 365;
 const net = slice - fees - tbill;
 return {
  capital_usd: capital,
  share,
  hold_days: holdDays,
  slice_usd: cents(slice),
  fees_usd: cents(fees),
  tbill_usd: cents(tbill),
  net_usd: cents(net),
  per_hour_usd: cents(net / hours),
  verdict: net > 0 ? 'FARM' : slice > fees ? 'THIN' : 'SKIP',
 };
}
export const VERDICTS = ['FARM', 'THIN', 'SKIP'];
export const VERDICT_TEXT = {
 FARM: 'The slice beats the fees and the T-bill yield given up',
 THIN: 'The slice beats the fees but not the T-bill yield',
 SKIP: 'The fees eat the slice',
};
/** Short, stable fingerprint of the ranking, so the page can show it is the board.json it ships with. */
export function boardDigest(rows) {
 let hash = 0x811c9dc5;
 for (const row of rows) {
  const text = `${row.slug}:${row.rank}:${row.implied_value_usd}|`;
  for (let i = 0; i < text.length; i += 1) {
   hash ^= text.charCodeAt(i);
   hash = Math.imul(hash, 0x01000193) >>> 0;
  }
 }
 return hash.toString(16).padStart(8, '0');
}
